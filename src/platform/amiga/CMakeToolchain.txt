# Toolchain file for cross-compiling to classic AmigaOS (m68k) using the
# AmigaPorts/Bebbo m68k-amigaos-gcc toolchain
# (https://github.com/AmigaPorts/m68k-amigaos-gcc).
#
# Usage:
#   cmake -B build-amiga -G Ninja \
#     -DCMAKE_TOOLCHAIN_FILE=src/platform/amiga/CMakeToolchain.txt \
#     -DAMIGA_TOOLCHAIN_PREFIX=/opt/m68k-amigaos-gcc-16.2 ..
#
# Target: 68040 with FPU. This is the best PiStorm/Emu68 target (Emu68's
# dynamic 68k->ARM translation models the 68040 most faithfully) and,
# unlike the 68060, the full 68040 has its complete FPU -- including the
# transcendentals (FSIN/FCOS/FETOX/...) the 68060 removed from silicon --
# so there's no FPSP-software-emulation story to worry about for real
# hardware either. Setting CMAKE_SYSTEM_NAME to "Generic" makes the
# top-level CMakeLists.txt automatically force DISABLE_FRONTENDS,
# MINIMAL_CORE, and a static build, same as the other embedded console
# ports (3DS/Wii/Switch/Vita).

if(DEFINED ENV{AMIGA_TOOLCHAIN_PREFIX} AND NOT DEFINED AMIGA_TOOLCHAIN_PREFIX)
	set(AMIGA_TOOLCHAIN_PREFIX $ENV{AMIGA_TOOLCHAIN_PREFIX})
endif()
if(NOT DEFINED AMIGA_TOOLCHAIN_PREFIX)
	set(AMIGA_TOOLCHAIN_PREFIX /opt/m68k-amigaos-gcc-16.2)
endif()

set(CMAKE_SYSTEM_NAME Generic CACHE INTERNAL "system name")
set(CMAKE_SYSTEM_PROCESSOR m68k CACHE INTERNAL "processor")

set(CMAKE_C_COMPILER "${AMIGA_TOOLCHAIN_PREFIX}/bin/m68k-amigaos-gcc" CACHE INTERNAL "c compiler")
set(CMAKE_CXX_COMPILER "${AMIGA_TOOLCHAIN_PREFIX}/bin/m68k-amigaos-g++" CACHE INTERNAL "cxx compiler")
set(CMAKE_ASM_COMPILER "${AMIGA_TOOLCHAIN_PREFIX}/bin/m68k-amigaos-gcc" CACHE INTERNAL "assembler")
set(CMAKE_AR "${AMIGA_TOOLCHAIN_PREFIX}/bin/m68k-amigaos-ar" CACHE INTERNAL "archiver")
set(CMAKE_RANLIB "${AMIGA_TOOLCHAIN_PREFIX}/bin/m68k-amigaos-ranlib" CACHE INTERNAL "ranlib")

# -D__BIG_ENDIAN__ is required: this compiler is big-endian by default
# (__BYTE_ORDER__ == __ORDER_BIG_ENDIAN__) but doesn't define the BSD-style
# __BIG_ENDIAN__ macro that include/mgba-util/common.h checks for.
#
# _GNU_SOURCE unlocks strdup/strndup/etc. in string.h, which strict
# -std=c11 otherwise hides behind feature-test macros -- same fix the Wii
# port applies for the same reason.
#
# -mcrt=newlib: on this specific toolchain build (GCC 16.2,
# AmigaPorts/m68k-amigaos-gcc), newlib's AmigaOS backing is complete
# (mkdir/getcwd/gettimeofday/fstat/dirent-with-d_type/getopt_long/full
# newlocale family all verified present and linkable) -- unlike the much
# older GCC 6.5.0b toolchain, where newlib lacked these and clib2/libnix
# each had their own gaps (clib2: no newlocale family, no dirent d_type,
# no getopt_long, plus a locale_t typedef conflict when mixed with
# newlib's own headers under _GNU_SOURCE; libnix: resident-library
# globals like __errno need separate wiring). newlib is also already
# this toolchain's default when -mcrt= is omitted; naming it explicitly
# just guards against that default ever changing.
#
# -fno-lto: on this (locally built, ad-hoc code-signed) macOS host,
# liblto_plugin.so fails to dlopen under Gatekeeper/hardened runtime even
# once the compiler/as/ld/collect2 binaries themselves have been approved,
# since library code-signing is enforced differently from launching an
# executable. Nothing here needs LTO; just avoid loading the plugin.
set(AMIGA_ARCH_FLAGS "-m68040 -mhard-float -mcrt=newlib -fno-lto -D__BIG_ENDIAN__ -D_GNU_SOURCE" CACHE INTERNAL "arch flags")

set(CMAKE_C_FLAGS "${AMIGA_ARCH_FLAGS}" CACHE INTERNAL "c compiler flags")
set(CMAKE_CXX_FLAGS "${AMIGA_ARCH_FLAGS}" CACHE INTERNAL "cxx compiler flags")
set(CMAKE_ASM_FLAGS "${AMIGA_ARCH_FLAGS}" CACHE INTERNAL "assembler flags")

set(CMAKE_FIND_ROOT_PATH "${AMIGA_TOOLCHAIN_PREFIX}")
set(CMAKE_FIND_ROOT_PATH_MODE_PROGRAM NEVER)
set(CMAKE_FIND_ROOT_PATH_MODE_LIBRARY ONLY)
set(CMAKE_FIND_ROOT_PATH_MODE_INCLUDE ONLY)

# Without this, find_package(ZLIB)/find_package(PNG) happily pick up the
# host macOS/Linux pkg-config entries instead of failing, which is worse
# than just not finding them.
set(PKG_CONFIG_EXECUTABLE "/dev/null" CACHE INTERNAL "" FORCE)

# This toolchain ships prebuilt static zlib/libpng archives matching the
# bundled headers -- simpler to link against directly than a shared
# .library route (z.library/png16_lib), which can still be added later
# if a smaller binary or a shared system copy becomes worth the extra
# complexity. Pre-seed the exact variant matching our -m68040
# -mhard-float flags (libm020/libm881 -- still the right variant for the
# 68040, same as the 68060, since both are 68020+ with a real FPU) so
# find_package(ZLIB)/
# find_package(PNG) (invoked via find_feature() in the top-level
# CMakeLists.txt) use these instead of searching.
set(ZLIB_INCLUDE_DIR "${AMIGA_TOOLCHAIN_PREFIX}/m68k-amigaos/include" CACHE PATH "")
set(ZLIB_LIBRARY "${AMIGA_TOOLCHAIN_PREFIX}/m68k-amigaos/lib/libm020/libm881/libz.a" CACHE FILEPATH "")
set(PNG_PNG_INCLUDE_DIR "${AMIGA_TOOLCHAIN_PREFIX}/m68k-amigaos/include" CACHE PATH "")
set(PNG_LIBRARY "${AMIGA_TOOLCHAIN_PREFIX}/m68k-amigaos/lib/libm020/libm881/libpng.a" CACHE FILEPATH "")

set(AMIGA ON)
add_definitions(-DAMIGA)

# CMake's own compiler-sanity-check links a trivial test executable before
# any of our targets exist. Restricting the initial probe to compiling
# only is the standard fix other embedded/Generic toolchain files use.
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)

# find_function()'s detection (src/platform/cmake/FindFunction.cmake) relies
# on -Wl,--require-defined try_compiles that don't reliably work for this
# Generic cross target. popcount32, snprintf_l, crc32, futimens, futimes,
# realpath all misdetect (true) under newlib/GCC 16.2 despite genuinely
# being absent -- forced off below so mgba's own fallbacks are used
# instead. If a similar misdetection resurfaces elsewhere, pre-seed the
# cache the same way: set(HAVE_X "" CACHE INTERNAL "") to force off, or
# add_definitions(-DHAVE_X) to force on.
#
# fsync/ftruncate have no HAVE_* guard upstream (called unconditionally
# from src/util/vfs/vfs-fd.c) and are also absent from newlib on this
# toolchain; see src/platform/amiga/startup.c for the shims that cover
# those two instead.
set(HAVE_POPCOUNT32 "" CACHE INTERNAL "")
set(HAVE_SNPRINTF_L "" CACHE INTERNAL "")
set(HAVE_CRC32 "" CACHE INTERNAL "")
set(HAVE_FUTIMENS "" CACHE INTERNAL "")
set(HAVE_FUTIMES "" CACHE INTERNAL "")
set(HAVE_REALPATH "" CACHE INTERNAL "")
